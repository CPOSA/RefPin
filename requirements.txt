# PinRef 依赖清单
# 安装：pip install -r requirements.txt
#
# 版本写法说明：>=X 是"至少要这个版本"，<Y 是"别升到会破坏兼容的大版本"。
# 这样既能装到新版修复，又不会哪天大版本更新把项目搞崩。

# --- 阶段 1 起就需要 ---

# 界面 / 窗口：置顶、无边框浮窗靠它
# 6.10 起才有 Python 3.14 的官方 wheel
PySide6>=6.10,<7

# 屏幕抓取：框选区域截图
mss>=10.0,<11

# --- 阶段 2 调色需要（阶段 1 可以先不用，但一起装省事） ---

# 像素级数学运算：灰度、色相、透明度都靠它向量化计算
numpy>=2.3,<3

# 图像读写与格式转换，配合 numpy 在 QPixmap 和数组之间来回转
Pillow>=11.0,<13

# --- 仅 macOS ---

# 用来把 app 切成 accessory 类型。不切的话，遇到有全屏 app 的屏幕，
# 遮罩根本不显示、那块屏就截不了（详见 README 第 7 节）。
# 末尾的 ; sys_platform == "darwin" 表示只在 macOS 上装，Windows 会自动跳过。
pyobjc-framework-Cocoa>=10,<13; sys_platform == "darwin"

# --- 阶段 3 以后可能用到，等做到那步再解开注释 ---
